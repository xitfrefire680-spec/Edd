local function GroundMove(targetXZ, speed, easeOut)
local _, hrp = getChar() if not hrp or not targetXZ then return false end
local groundY = hrp.Position.Y
local goal = Vector3.new(targetXZ.X, groundY, targetXZ.Z)
local start = hrp.Position
local dx = goal.X - start.X local dz = goal.Z - start.Z
local dist = math.sqrt(dx*dx + dz*dz)
if dist < 1 then hrp.CFrame = CFrame.new(goal) hrp.AssemblyLinearVelocity = Vector3.zero hrp.AssemblyAngularVelocity = Vector3.zero return true end
speed = math.clamp(tonumber(speed) or 750, 50, 40000)
local t0 = os.clock() local maxTime = (dist / math.max(speed, 50)) * 2 + 4
local cx, cz = hrp.Position.X, hrp.Position.Z
-- 🔥 OTIMIZAÇÃO: SpoofState a cada 3 frames
local spoofTick = 0
while not HUB.dead do
local dt = RunService.Heartbeat:Wait()
local tx = goal.X - cx local tz = goal.Z - cz
local rem = math.sqrt(tx*tx + tz*tz) if rem < 1 then break end
local sp2 = speed
if easeOut then local prog = 1 - math.clamp(rem / dist, 0, 1) sp2 = math.max(speed * (1 - prog * 0.8), 35) end
local step = math.min(sp2 * dt, rem)
local ux, uz = tx / rem, tz / rem
cx = cx + ux * step cz = cz + uz * step
local np = Vector3.new(cx, groundY, cz) local lookAt = Vector3.new(cx + ux, groundY, cz + uz)
hrp.CFrame = CFrame.lookAt(np, lookAt)
local realVel = step / math.max(dt, 0.001)
hrp.AssemblyLinearVelocity = Vector3.new(ux * realVel, 0, uz * realVel)
hrp.AssemblyAngularVelocity = Vector3.zero
spoofTick = spoofTick + 1
if spoofTick % 3 == 0 then
    SetSpoofWalkSpeed(HUB.flySpeed)
    if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
    SyncHoverPad()
end
if os.clock() - t0 > maxTime then break end
end
hrp.CFrame = CFrame.new(goal) hrp.AssemblyLinearVelocity = Vector3.zero hrp.AssemblyAngularVelocity = Vector3.zero
return true
end

local function returnHome(flyBackPos)
    if not isCarryingEgg() then return false end

    local dest = getPlot1Pos()

    end

end

local function runCycle()
    if HUB.busy or HUB.dead or not HUB.running then return end
    HUB.busy = true
    if isCarryingEgg() then returnHome() HUB.busy = false return end

    if HUB.activeTarget and HUB.activeTarget.uid then
        local rec = findRecord(HUB.activeTarget.uid)
        if rec then
            local _, hrpBefore = getChar()
            local flyPos = hrpBefore and hrpBefore.Position
            if stealEgg(rec) then
                HUB.activeTarget = nil
                returnHome(flyPos)
            end
            HUB.busy = false return
        else HUB.activeTarget = nil end
    end

    local eggs = getFieldEggs()
    if #eggs == 0 then HUB.busy = false return end

    local target = nil
    if HUB.selectedUid then
        for _, e in ipairs(eggs) do if e.uid == HUB.selectedUid then target = e break end end
    end
    if not target then
        table.sort(eggs, function(a, b) if a.value ~= b.value then return a.value > b.value end return a.rank > b.rank end)
        target = eggs[1]
    end

    local _, hrpB = getChar()
    local flyPosB = hrpB and hrpB.Position
    HUB.activeTarget = { uid = target.uid, name = target.name }
    if stealEgg(target.record) then
        HUB.activeTarget = nil
        returnHome(flyPosB)
    end
    HUB.busy = false
end

local ChickenBossFarm = {
    enabled = false, stage = "idle",
    chickenName = "Chicken",
    altNames = {"galinha","hen","rooster","chicken"},
    hitThreshold = 2, lastHealth = 100,
    stageStartTime = 0, waitTimeout = 120,
}
local function findChicken()
    local hints = {}
    for _, n in ipairs(ChickenBossFarm.altNames) do table.insert(hints, n:lower()) end
    table.insert(hints, ChickenBossFarm.chickenName:lower())
    local candidates = {}
    local function scan(container, depth)
        if depth > 6 or not container then return end
        for _, obj in ipairs(container:GetChildren()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                local n = obj.Name:lower()
                for _, hint in ipairs(hints) do
                    if n:find(hint, 1, true) then
                        local pos
                        if obj:IsA("BasePart") then pos = obj.Position
                        else local prim = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart") if prim then pos = prim.Position end end
                        if pos then table.insert(candidates, {obj=obj, pos=pos}) end
                        break
                    end
                end
            end
            if obj:IsA("Model") or obj:IsA("Folder") then scan(obj, depth + 1) end
        end
    end
    scan(workspace, 0)
    local _, hrp = getChar()
    if hrp and #candidates > 1 then table.sort(candidates, function(a,b) return (a.pos-hrp.Position).Magnitude < (b.pos-hrp.Position).Magnitude end) end
    return candidates[1]
end
local function grabChickenEgg(chickenObj)
    if not chickenObj then return false end
    for _, d in ipairs(chickenObj:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled then pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end) end
    end
    local _, hrp = getChar()
    if hrp then
        local anchor = chickenObj:IsA("BasePart") and chickenObj or chickenObj:FindFirstChildWhichIsA("BasePart")
        if anchor then pcall(function() firetouchinterest(hrp, anchor, 0) firetouchinterest(hrp, anchor, 1) end) end
    end
    return true
end
local function RunChickenSequence()
    if not ChickenBossFarm.enabled or ChickenBossFarm.stage == "done" then return end
    if HUB.dead then return end
    if ChickenBossFarm.stage == "idle" then
        ChickenBossFarm.stage = "going_chicken"
        if setStatus then setStatus("🐔 Indo na galinha...", true) end
        local chicken = findChicken()
        if not chicken then
            if setStatus then setStatus("❌ Galinha não achada (pulando)", false) end
            ChickenBossFarm.stage = "done"
            return
        end
        flyTo(chicken.pos, 15)
        local chicken2 = findChicken()
        if chicken2 then grabChickenEgg(chicken2.obj) end
        ChickenBossFarm.lastHealth = getChar() and getChar().Health or 100
        ChickenBossFarm.stage = "waiting_hit"
        ChickenBossFarm.stageStartTime = tick()
        if setStatus then setStatus("⏳ Esperando boss bater...", true) end
        return
    end
    if ChickenBossFarm.stage == "waiting_hit" then
        if tick() - ChickenBossFarm.stageStartTime > ChickenBossFarm.waitTimeout then
            if setStatus then setStatus("⏱️ Timeout (pulando)", false) end
            ChickenBossFarm.stage = "done"
        end
        return
    end
end

local function SingleClickPickup(uid, eggName)
    if HUB.dead then return end
    if HUB.activeTarget and HUB.activeTarget.uid == uid then return end
    HUB.activeTarget = { uid = uid, name = eggName, priority = true }
    if setStatus then setStatus("🎯 Pegando " .. eggName .. "...", true) end
    if not HUB.running then
        task.spawn(function()
            local waited = 0
            while HUB.busy and waited < 10 do
                RunService.Heartbeat:Wait()
                waited = waited + 0.1
            end
            if HUB.dead then HUB.activeTarget = nil return end
            HUB.busy = true
            local success = false
            local _, hrpPre = getChar()
            local flyPos = hrpPre and hrpPre.Position
            pcall(function()
                local rec = findRecord(uid)
                if rec then success = stealEgg(rec) end
            end)
            if success then
                pcall(function() returnHome(flyPos) end)
            else
                if setStatus then setStatus("❌ Não conseguiu pegar", false) end
            end
            HUB.activeTarget = nil
            HUB.busy = false
        end)
    end
end

local function refreshEggList()
local eggs = getFieldEggs()
if HUB.sortMode=="GEN" then table.sort(eggs, function(a,b) if a.value~=b.value then return a.value>b.value end return a.rank>b.rank end)
else table.sort(eggs, function(a,b) if a.rank~=b.rank then return a.rank>b.rank end return a.value>b.value end) end
local seen = {} local newIcons = {}
for i, e in ipairs(eggs) do
seen[e.uid] = true
local r = rowMap[e.uid]
if not r then r = createRow(e, i) if e.icon then newIcons[#newIcons+1] = e.icon end
else
r.frame.LayoutOrder = i
local genText = string.format("Gen %s    KG %s    📍 %s", fmtValue(e.value), fmtKg(e.kg), e.area)
if r.lastGenText ~= genText then r.genLbl.Text = genText r.lastGenText = genText end
local rarText = "Rarity " .. e.rarity local rarColor = e.rcolor or RARITY_COLOR[e.rarity] or THEME.textMuted
if r.lastRarText ~= rarText then r.rarLbl.Text = rarText r.lastRarText = rarText end
if r.lastRarColor ~= rarColor then r.rarLbl.TextColor3 = rarColor r.lastRarColor = rarColor end
if r.nameLbl.Text ~= e.name then r.nameLbl.Text = e.name end
if e.icon and r.photo.Image ~= e.icon then r.photo.Image = e.icon newIcons[#newIcons+1] = e.icon end
local picked = (HUB.selectedUid == e.uid)
if picked ~= r.isPicked then r.isPicked = picked r.frame.BackgroundColor3 = picked and THEME.rowPicked or THEME.rowNormal end
end
end

local function findAdminBelt()

        if f and f:IsA("BasePart") then return f end
    end
    for _, o in ipairs(at:GetDescendants()) do

    end
    return false
end

local AutoHitSystem = {
    lastHit = 0,
    cooldown = 0.35,
    busy = false,
    currentTarget = nil,
    swings = 0,
}

local function findEnemyCarrier()
    if not EggState or not EggState.ReadFieldEggs then return nil end
    local ok, snap = pcall(EggState.ReadFieldEggs)
    if not ok or not snap or not snap.Records then return nil end
    local best, bestVal = nil, -1
    for _, r in ipairs(snap.Records) do
        if r.State == "Carried" and r.CarrierUserId and r.CarrierUserId ~= LP.UserId then
            local carrier = getPlayerByUserId(r.CarrierUserId)
            if carrier and getPlayerHrp(carrier) then
                local cat = r.AssetCategory or r.Category
                local val = getPetValue(cat)
                if val > bestVal then
                    bestVal = val
                    best = { record = r, carrier = carrier, uid = r.Uid, value = val }
                end
            end
        end
    end
    return best
end

local function equipBestWeapon()
    local char = LP.Character
    if not char then return nil end
    local isWeapon = function(tool)
        local n = string.lower(tool.Name)
        return n:find("bat") or n:find("sword") or n:find("hammer")
            or n:find("axe") or n:find("blade") or n:find("staff")
            or n:find("club") or n:find("katana")
    end
    local equipped = char:FindFirstChildOfClass("Tool")
    if equipped and isWeapon(equipped) then return equipped end
    local bp = LP:FindFirstChildOfClass("Backpack")
    if not bp then return nil end
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") and isWeapon(t) then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                pcall(function() hum:EquipTool(t) end)
                task.wait(0.08)
                local eq = char:FindFirstChildOfClass("Tool")
                if eq then return eq end
            end
        end
    end
    return nil
end

local function runAutoHit()
    if AutoHitSystem.busy then return end
    if HUB.dead then return end
    if not HUB.hitEnabled then return end
    if os.clock() - AutoHitSystem.lastHit < AutoHitSystem.cooldown then return end

    local target = findEnemyCarrier()
    if not target then
        AutoHitSystem.currentTarget = nil
        return
    end

    AutoHitSystem.busy = true

    local cHrp = getPlayerHrp(target.carrier)
    local _, myHrp = getChar()
    if not cHrp or not myHrp then
        AutoHitSystem.busy = false
        return
    end

    AutoHitSystem.currentTarget = target.carrier.Name

    local dist = (myHrp.Position - cHrp.Position).Magnitude

    if dist > 12 then
        local moveTo = cHrp.Position + cHrp.CFrame.LookVector * 3 + Vector3.new(0, 2, 0)
        myHrp.CFrame = CFrame.new(moveTo)
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
        SetSpoofWalkSpeed(HUB.flySpeed)
        if BypassState.integrityState then
            SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM)
        end
        SyncHoverPad()
        task.wait(0.08)
    end

    pcall(function()
        myHrp.CFrame = CFrame.lookAt(
            myHrp.Position,
            Vector3.new(cHrp.Position.X, myHrp.Position.Y, cHrp.Position.Z)
        )
    end)

    local weapon = equipBestWeapon()
    if weapon then
        pcall(function() weapon:Activate() end)
        AutoHitSystem.lastHit = os.clock()
        AutoHitSystem.swings = AutoHitSystem.swings + 1
        if setStatus then
            setStatus("⚔️ Batendo em " .. target.carrier.Name ..
                " (R$" .. fmtValue(target.value) .. ") · " .. AutoHitSystem.swings .. " swings", true)
        end
    else
        if setStatus then setStatus("❌ Sem arma no inventário", false) end
    end

    AutoHitSystem.busy = false
end

local ChickenBossFarm = {
    enabled = false, stage = "idle",
    chickenName = "Chicken",
    altNames = {"galinha","hen","rooster","chicken"},
    hitThreshold = 2, lastHealth = 100,
    stageStartTime = 0, waitTimeout = 120,
}
local function findChicken()
    local hints = {}
    for _, n in ipairs(ChickenBossFarm.altNames) do table.insert(hints, n:lower()) end
    table.insert(hints, ChickenBossFarm.chickenName:lower())
    local candidates = {}
    local function scan(container, depth)
        if depth > 6 or not container then return end
        for _, obj in ipairs(container:GetChildren()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                local n = obj.Name:lower()
                for _, hint in ipairs(hints) do
                    if n:find(hint, 1, true) then
                        local pos
                        if obj:IsA("BasePart") then pos = obj.Position
                        else local prim = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart") if prim then pos = prim.Position end end
                        if pos then table.insert(candidates, {obj=obj, pos=pos}) end
                        break
                    end
                end
            end
            if obj:IsA("Model") or obj:IsA("Folder") then scan(obj, depth + 1) end
        end
    end
    scan(workspace, 0)
    local _, hrp = getChar()
    if hrp and #candidates > 1 then table.sort(candidates, function(a,b) return (a.pos-hrp.Position).Magnitude < (b.pos-hrp.Position).Magnitude end) end
    return candidates[1]
end
local function grabChickenEgg(chickenObj)
    if not chickenObj then return false end
    for _, d in ipairs(chickenObj:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled then pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end) end
    end
    local _, hrp = getChar()
    if hrp then
        local anchor = chickenObj:IsA("BasePart") and chickenObj or chickenObj:FindFirstChildWhichIsA("BasePart")
        if anchor then pcall(function() firetouchinterest(hrp, anchor, 0) firetouchinterest(hrp, anchor, 1) end) end
    end
    return true
end
local function RunChickenSequence()
    if not ChickenBossFarm.enabled or ChickenBossFarm.stage == "done" then return end
    if HUB.dead then return end
    if ChickenBossFarm.stage == "idle" then
        ChickenBossFarm.stage = "going_chicken"
        if setStatus then setStatus("🐔 Indo na galinha...", true) end
        local chicken = findChicken()
        if not chicken then
            if setStatus then setStatus("❌ Galinha não achada (pulando)", false) end
            ChickenBossFarm.stage = "done"
            return
        end
        flyTo(chicken.pos, 15)
        local chicken2 = findChicken()
        if chicken2 then grabChickenEgg(chicken2.obj) end
        ChickenBossFarm.lastHealth = getChar() and getChar().Health or 100
        ChickenBossFarm.stage = "waiting_hit"
        ChickenBossFarm.stageStartTime = tick()
        if setStatus then setStatus("⏳ Esperando boss bater...", true) end
        return
    end
    if ChickenBossFarm.stage == "waiting_hit" then
        if tick() - ChickenBossFarm.stageStartTime > ChickenBossFarm.waitTimeout then
            if setStatus then setStatus("⏱️ Timeout (pulando)", false) end
            ChickenBossFarm.stage = "done"
        end
        return
    end
end
